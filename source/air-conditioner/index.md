<script>
(function(){
  var root=document.getElementById('mteacher-ac');
  if(!root||root.dataset.init)return; root.dataset.init='1';
  var $=function(id){return document.getElementById(id);};
  var S={power:false,mode:'cool',temp:26,fan:'auto',swing:true,timer:0,room:33.0,kwh:0};
  try{var sv=JSON.parse(localStorage.getItem('mteacher_ac')||'{}');['mode','temp','fan','swing'].forEach(function(k){if(sv[k]!==undefined)S[k]=sv[k];});}catch(e){}
  function save(){try{localStorage.setItem('mteacher_ac',JSON.stringify({mode:S.mode,temp:S.temp,fan:S.fan,swing:S.swing}));}catch(e){}}
  var QUIPS={
    on:['清凉配送中，请签收 ❄','空调已上岗，开始打工！','呼——第一缕凉风，敬自由。','制冷模块加载完毕，快乐+1。'],
    cool:['再低一度，北极熊给你点赞 🐻‍❄️','26°C 是体感与电费的黄金分割点。','凉风有信，秋月无边，电费有数。'],
    heat:['暖气已就位，秋裤可以退役了。','制热模式：把夏天揣进口袋。'],
    off:['空调下班了，它去吹别人的风了。','已关机，真省电，马老师欣慰。','定时到了，空调说它要午休。'],
    low:['16°C？你是想在客厅看极光吗？','30°C？这是在给空调放年假。']
  };
  function pick(a){return a[Math.floor(Math.random()*a.length)];}
  function msg(t){$('ac-msg').textContent=t;}
  var MODEC={cool:'#0ea5e9',heat:'#f97316',dry:'#14b8a6',fan:'#65a30d'};
  var MODEW={cool:'#7dd3fc',heat:'#fdba74',dry:'#5eead4',fan:'#bef264'};
  var FANS={auto:['自动风',1],low:['低速',0.6],mid:['中速',1],high:['高速',1.6]};
  var KW={cool:1.1,heat:1.4,dry:0.8,fan:0.05};
  function fmtT(s){var h=Math.floor(s/3600),m=Math.floor(s%3600/60);return h+'h '+m+'m';}
  function render(){
    root.classList.toggle('on',S.power);
    root.classList.toggle('locked',!S.power);
    $('ac-temp').textContent=S.power?S.temp:'--';
    $('ac-set').textContent=S.temp;
    $('ac-dot').style.background=S.power?MODEC[S.mode]:'#cbd5e1';
    $('ac-dot').style.boxShadow=S.power?('0 0 12px '+MODEC[S.mode]):'none';
    $('ac-room').textContent=S.room.toFixed(1)+'°C';
    $('ac-timer').textContent=S.timer>0?('剩余 '+fmtT(S.timer)):'未定时';
    $('ac-fanlabel').textContent=FANS[S.fan][0];
    $('ac-louver').classList.toggle('swing',S.power&&S.swing);
    $('ac-swing').textContent=S.swing?'开':'关';
    $('ac-swing').classList.toggle('sel',S.swing);
    var btns=root.querySelectorAll('.ac-mode');
    btns.forEach(function(b){b.className='ac-btn ac-mode'+(S.power&&b.dataset.mode===S.mode?' sel-'+b.dataset.mode:'');});
    root.querySelectorAll('[data-fan]').forEach(function(b){b.classList.toggle('sel',b.dataset.fan===S.fan);});
    root.querySelectorAll('[data-timer]').forEach(function(b){b.classList.toggle('sel',+b.dataset.timer===(S._timerSet||0));});
    var winds=$('ac-wind').children;
    var dur={auto:1.6,low:2.4,mid:1.4,high:0.8}[S.fan];
    for(var i=0;i<winds.length;i++){
      var w=winds[i];
      w.style.left=(6+i*13)+'%';
      w.style.height=(46+((i*37)%40))+'px';
      w.style.background=MODEW[S.mode];
      w.style.animationDuration=dur+'s';
      w.style.animationDelay=(i*0.28)+'s';
    }
    $('ac-bill').textContent='虚拟电费 ¥'+(S.kwh*0.6).toFixed(2)+'（本页电费由马老师的心意支付）';
  }
  function needPower(){if(!S.power){msg('💤 先按电源键开机呀，空调还没睡醒。');return false;}return true;}
  $('ac-power').onclick=function(){
    S.power=!S.power;
    if(S.power){S.timer=0;S._timerSet=0;msg('🎉 '+pick(QUIPS.on));}
    else{S.timer=0;S._timerSet=0;msg('👋 '+pick(QUIPS.off));}
    render();
  };
  root.querySelectorAll('.ac-mode').forEach(function(b){b.onclick=function(){
    if(!needPower())return;
    S.mode=b.dataset.mode;save();
    msg({cool:'❄ '+pick(QUIPS.cool),heat:'☀ '+pick(QUIPS.heat),dry:'💧 除湿模式：空气都变干爽了（并没有）。',fan:'🌀 送风模式：纯风，零制冷，快乐翻倍。'}[S.mode]);
    render();
  };});
  $('ac-tup').onclick=function(){if(!needPower())return;if(S.temp<30){S.temp++;save();}msg(S.temp>=30?'🔥 '+pick(QUIPS.low):('🌡️ 设定 '+S.temp+'°C，'+(S.temp<=20?'勇士。':'刚刚好。')));render();};
  $('ac-tdown').onclick=function(){if(!needPower())return;if(S.temp>16){S.temp--;save();}msg(S.temp<=16?'🧊 '+pick(QUIPS.low):('🌡️ 设定 '+S.temp+'°C，'+(S.temp<=20?'勇士。':'刚刚好。')));render();};
  root.querySelectorAll('[data-fan]').forEach(function(b){b.onclick=function(){if(!needPower())return;S.fan=b.dataset.fan;save();msg('💨 风速：'+FANS[S.fan][0]+(S.fan==='high'?'，头发预警！':''));render();};});
  $('ac-swing').onclick=function(){if(!needPower())return;S.swing=!S.swing;save();msg(S.swing?'↔️ 摆风开启，雨露均沾。':'⏸️ 摆风关闭，定点投送。');render();};
  root.querySelectorAll('[data-timer]').forEach(function(b){b.onclick=function(){
    if(!needPower())return;
    S._timerSet=+b.dataset.timer;S.timer=S._timerSet;save();
    msg(S.timer>0?('⏱️ 已定时 '+fmtT(S.timer)+' 后自动关机，放心睡。'):'⏱️ 定时已取消，空调表示可以加班。');
    render();
  };});
  // 雪花/热浪粒子
  var scene=$('ac-scene');
  setInterval(function(){
    if(!S.power)return;
    if(scene.querySelectorAll('.ac-snow').length>26)return;
    var p=document.createElement('span');
    p.className='ac-snow';
    p.textContent=S.mode==='heat'?'♨':(S.mode==='dry'?'💧':'❄');
    p.style.left=(Math.random()*92+3)+'%';
    p.style.fontSize=(10+Math.random()*10)+'px';
    p.style.animationDuration=(1.6+Math.random()*1.6)+'s';
    scene.appendChild(p);
    setTimeout(function(){p.remove();},3400);
  },420);
  // 每秒：室温漂移 + 电费 + 定时
  setInterval(function(){
    if(!S.power)return;
    var f=FANS[S.fan][1];
    var target=S.mode==='cool'?S.temp:(S.mode==='heat'?S.temp:(S.mode==='dry'?26:S.room));
    var rate=(S.mode==='fan'?0:0.022*f);
    var d=target-S.room;
    if(Math.abs(d)>0.005)S.room+=Math.max(-rate,Math.min(rate,d));
    S.kwh+=KW[S.mode]*(0.5+0.5*f)/3600;
    if(S.timer>0){S.timer--;if(S.timer<=0){S.power=false;S._timerSet=0;msg('👋 '+QUIPS.off[2]);}}
    render();
  },1000);
  render();
  // pjax 兼容
  document.addEventListener('pjax:complete',function(){delete root.dataset.init;});
})();
</script>
